def clear_scene():
    bpy.ops.object.select_all(action="SELECT")
    bpy.ops.object.delete(use_global=False)
    for mesh in list(bpy.data.meshes):
        bpy.data.meshes.remove(mesh)
    for mat in list(bpy.data.materials):
        bpy.data.materials.remove(mat)


def _tex_uv_chain(nt, texture_path):
    tex_coord = nt.nodes.new("ShaderNodeTexCoord"); tex_coord.location = (-600, 100)
    mapping   = nt.nodes.new("ShaderNodeMapping");   mapping.location   = (-400, 100)
    img_tex   = nt.nodes.new("ShaderNodeTexImage");  img_tex.location   = (-150, 100)
    nt.links.new(tex_coord.outputs["UV"], mapping.inputs["Vector"])
    nt.links.new(mapping.outputs["Vector"], img_tex.inputs["Vector"])
    if texture_path and os.path.exists(texture_path):
        try:
            img_tex.image = bpy.data.images.load(texture_path, check_existing=True)
            return img_tex, True
        except Exception as e:
            print(f"    ⚠️  Texture load failed: {e}")
    return img_tex, False


def _uv_noise(nt, scale, detail, roughness, distortion, offset_x=0, offset_y=0):
    tc = nt.nodes.new("ShaderNodeTexCoord"); tc.location = (offset_x - 400, offset_y)
    mp = nt.nodes.new("ShaderNodeMapping");   mp.location = (offset_x - 200, offset_y)
    ns = nt.nodes.new("ShaderNodeTexNoise");  ns.location = (offset_x,       offset_y)
    ns.inputs["Scale"].default_value      = scale
    ns.inputs["Detail"].default_value     = detail
    ns.inputs["Roughness"].default_value  = roughness
    ns.inputs["Distortion"].default_value = distortion
    nt.links.new(tc.outputs["UV"], mp.inputs["Vector"])
    nt.links.new(mp.outputs["Vector"], ns.inputs["Vector"])
    return ns

def make_planet_material(name, texture_path, base_color):
    roughness, metallic, emission_strength, has_atmo = PLANET_PRESETS.get(
        name, (0.80, 0.00, 0.80, False))

    mat_name = name + "_Mat" if not name.endswith("_Mat") else name
    mat = bpy.data.materials.new(name=mat_name)
    mat.use_nodes = True
    nt = mat.node_tree
    nt.nodes.clear()

    output   = nt.nodes.new("ShaderNodeOutputMaterial"); output.location   = (1000, 0)
    add_sh   = nt.nodes.new("ShaderNodeAddShader");      add_sh.location   = (780,  0)
    bsdf     = nt.nodes.new("ShaderNodeBsdfPrincipled"); bsdf.location     = (480,  80)
    bsdf.inputs["Roughness"].default_value = roughness
    bsdf.inputs["Metallic"].default_value  = metallic
    emission = nt.nodes.new("ShaderNodeEmission");       emission.location = (480, -120)
    emission.inputs["Strength"].default_value = emission_strength

    img_tex, loaded = _tex_uv_chain(nt, texture_path)

    if loaded:
        color_source = img_tex.outputs["Color"]
    else:
        color_source = _make_procedural_planet(nt, name, base_color)

    nt.links.new(color_source,              bsdf.inputs["Base Color"])
    nt.links.new(color_source,              emission.inputs["Color"])
    nt.links.new(bsdf.outputs["BSDF"],      add_sh.inputs[0])
    nt.links.new(emission.outputs["Emission"], add_sh.inputs[1])

    if has_atmo:
        atmo_emission = _add_atmosphere_rim(nt, name)
        add_sh2 = nt.nodes.new("ShaderNodeAddShader"); add_sh2.location = (900, 0)
        nt.links.new(add_sh.outputs["Shader"],         add_sh2.inputs[0])
        nt.links.new(atmo_emission.outputs["Emission"], add_sh2.inputs[1])
        nt.links.new(add_sh2.outputs["Shader"],        output.inputs["Surface"])
    else:
        nt.links.new(add_sh.outputs["Shader"], output.inputs["Surface"])

    return mat

def _add_atmosphere_rim(nt, name):
    atmo_colors = {
        "Earth":   (0.35, 0.60, 1.00),

        "Uranus":  (0.45, 0.85, 0.90),
        "Neptune": (0.25, 0.50, 1.00),

    atmo_col = atmo_colors.get(name, (0.6, 0.7, 1.0))
    fresnel = nt.nodes.new("ShaderNodeFresnel");  fresnel.location  = (250, -350)
    fresnel.inputs["IOR"].default_value = 1.05
    ramp    = nt.nodes.new("ShaderNodeValToRGB"); ramp.location      = (450, -350)

    emission.inputs["Color"].default_value    = (*atmo_col, 1.0)
    emission.inputs["Strength"].default_value = 0.6
    nt.links.new(fresnel.outputs["Fac"], ramp.inputs["Fac"])
    nt.links.new(ramp.outputs["Color"],  emission.inputs["Color"])
    return emission

def create_saturn_rings(parent_obj, planet_radius, texture_path=None):
    import bmesh

    scale = (planet_radius / 3.0) * (2.3 / 3.3)
    ring_specs = [
        (3.50 * scale, 4.00 * scale, "D_Ring", 0.00, 0.08, 0.30),
        (4.05 * scale, 5.20 * scale, "C_Ring", 0.08, 0.30, 0.55),
        (5.25 * scale, 7.40 * scale, "B_Ring", 0.30, 0.68, 0.95),
        (7.60 * scale, 9.50 * scale, "A_Ring", 0.68, 0.95, 0.90),
        (9.70 * scale, 9.90 * scale, "F_Ring", 0.95, 1.00, 0.45),
    ]

    ring_img = None
    if texture_path and os.path.exists(texture_path):
        ring_img = bpy.data.images.get("Saturn_Rings") or bpy.data.images.load(texture_path)
        ring_img.name = "Saturn_Rings"
        ring_img.colorspace_settings.name = "sRGB"

    ring_parent = bpy.data.objects.new("Saturn_RingSystem", None)
    bpy.context.collection.objects.link(ring_parent)
    ring_parent.empty_display_type = "PLAIN_AXES"
    ring_parent.empty_display_size = 0.01
    ring_parent.rotation_euler = (math.radians(26.7), 0.0, 0.0)
    ring_parent.parent = parent_obj

    SEGMENTS = 256
    for inner, outer, rname, u0, u1, alpha in ring_specs:
        bm = bmesh.new()
        uv_layer = bm.loops.layers.uv.new("UVMap")
        cos_a = [math.cos(2.0 * math.pi * i / SEGMENTS) for i in range(SEGMENTS)]
        sin_a = [math.sin(2.0 * math.pi * i / SEGMENTS) for i in range(SEGMENTS)]
        for i in range(SEGMENTS):
            j = (i + 1) % SEGMENTS
            vi0 = bm.verts.new((inner * cos_a[i], inner * sin_a[i], 0.0))
            vo0 = bm.verts.new((outer * cos_a[i], outer * sin_a[i], 0.0))
            vo1 = bm.verts.new((outer * cos_a[j], outer * sin_a[j], 0.0))
            vi1 = bm.verts.new((inner * cos_a[j], inner * sin_a[j], 0.0))
            face = bm.faces.new([vi0, vo0, vo1, vi1])
            t0 = i / SEGMENTS; t1 = (i + 1) / SEGMENTS
            face.loops[0][uv_layer].uv = (u0, t0)
            face.loops[1][uv_layer].uv = (u1, t0)
            face.loops[2][uv_layer].uv = (u1, t1)
            face.loops[3][uv_layer].uv = (u0, t1)
        mesh = bpy.data.meshes.new(rname + "_Mesh")
        bm.to_mesh(mesh); bm.free()
        mesh.uv_layers[0].name = "UVMap"
        ring = bpy.data.objects.new(rname, mesh)
        bpy.context.collection.objects.link(ring)
        ring.data.materials.append(
            make_ring_material(ring_img, alpha_fallback=alpha, ring_name=rname))
        bpy.context.view_layer.objects.active = ring
        ring.select_set(True)
        bpy.ops.object.shade_flat()
        ring.parent = ring_parent

    print("  ✅ Saturn rings built with 26.7° tilt")
    return ring_parent

def _spawn_body(parent_empty, mat, x, y, z, size_ranges, subdivisions=1):
    size = random.choice(size_ranges)
    bpy.ops.mesh.primitive_ico_sphere_add(
        radius=size, subdivisions=subdivisions, location=(x, y, z))
    obj = bpy.context.active_object
    obj.rotation_euler = (
        random.uniform(0, math.pi * 2),
        random.uniform(0, math.pi * 2),
        random.uniform(0, math.pi * 2),
    )
    sx = random.uniform(0.6, 1.4)
    sy = random.uniform(0.6, 1.4)
    sz = random.uniform(0.5, 1.2)
    obj.scale = (sx, sy, sz)
    bpy.ops.object.transform_apply(scale=True)
    obj.data.materials.append(mat)
    obj.parent = parent_empty
    return obj


def build_asteroid_belt(num_asteroids=500, inner_r=26.0, outer_r=31.5,
                        max_z=0.6, seed=42, num_rogues=20):
    print(f"\n  🪨 Building asteroid belt ({num_asteroids} rocks + {num_rogues} rogues)...")
    random.seed(seed)

    mat = make_asteroid_material()

    belt_empty = bpy.data.objects.new("AsteroidBelt", None)
    belt_empty.empty_display_type = "PLAIN_AXES"
    belt_empty.empty_display_size = 0.01
    bpy.context.collection.objects.link(belt_empty)

    mid_r = (inner_r + outer_r) / 2.0

    main_sizes = (
        [random.uniform(0.04, 0.09) for _ in range(40)] +
        [random.uniform(0.09, 0.16) for _ in range(25)] +
        [random.uniform(0.16, 0.26) for _ in range(8)]  +
        [random.uniform(0.26, 0.50) for _ in range(2)]
    )

    for i in range(num_asteroids):
        radius = random.triangular(inner_r, outer_r, mid_r)
        angle  = random.uniform(0.0, 2.0 * math.pi)
        x = math.cos(angle) * radius
        y = math.sin(angle) * radius
        z = random.gauss(0.0, max_z / 2.5)
        z = max(-max_z, min(max_z, z))
        obj = _spawn_body(belt_empty, mat, x, y, z, main_sizes)
        obj.name = f"Asteroid_{i:03d}"

    rogue_sizes = [random.uniform(0.05, 0.20) for _ in range(20)]
    rogue_span  = (outer_r - inner_r)

    for i in range(num_rogues):
        if random.random() < 0.5:
            radius = random.uniform(inner_r - rogue_span * 0.6, inner_r - 0.5)
        else:
            radius = random.uniform(outer_r + 0.5, outer_r + rogue_span * 0.6)
        radius = max(radius, 1.0)
        angle  = random.uniform(0.0, 2.0 * math.pi)
        x = math.cos(angle) * radius
        y = math.sin(angle) * radius
        z = random.gauss(0.0, max_z * 1.8)
        z = max(-max_z * 4.0, min(max_z * 4.0, z))
        obj = _spawn_body(belt_empty, mat, x, y, z, rogue_sizes)
        obj.name = f"Asteroid_Rogue_{i:02d}"

    print(f"  ✅ Asteroid belt — {num_asteroids} rocks (r={inner_r}-{outer_r}) + {num_rogues} rogues")
    return belt_empty

def animate_local_orbit(moon_tilt_empty, orbit_radius, period_frames, phase_deg=0.0):
    scene        = bpy.context.scene
    total_frames = scene.frame_end
    steps        = 64
    num_keys     = int(total_frames / period_frames * steps) + steps
    phase_rad    = math.radians(phase_deg)
    for k in range(num_keys + 1):
        frame = 1 + k * (period_frames / steps)
        if frame > total_frames:
            break
        angle = phase_rad + (2 * math.pi * k) / steps
        moon_tilt_empty.location = (
            math.cos(angle) * orbit_radius,
            math.sin(angle) * orbit_radius,
            0,
        )
        moon_tilt_empty.keyframe_insert(data_path="location", frame=frame)
    if moon_tilt_empty.animation_data and moon_tilt_empty.animation_data.action:
        for fcurve in moon_tilt_empty.animation_data.action.fcurves:
            for kf in fcurve.keyframe_points:
                kf.interpolation = "LINEAR"

def animate_pure_z_spin(obj, period_frames, direction=1):
    scene        = bpy.context.scene
    total_frames = scene.frame_end
    obj.rotation_mode  = "XYZ"
    obj.rotation_euler = (0.0, 0.0, 0.0)
    num_spins = max(1, math.ceil(total_frames / period_frames))
    for spin in range(num_spins + 1):
        frame = 1 + spin * period_frames
        obj.rotation_euler[2] = direction * 2 * math.pi * spin
        obj.keyframe_insert(data_path="rotation_euler", index=2, frame=frame)
    if obj.animation_data and obj.animation_data.action:
        for fcurve in obj.animation_data.action.fcurves:
            if fcurve.array_index == 2:
                for kf in fcurve.keyframe_points:
                    kf.interpolation = "LINEAR"
                fcurve.modifiers.new(type="CYCLES")

    return base_direction

    cam.data.lens     = 50
    cam.data.clip_end = 1000.0

    cam.data.dof.use_dof      = True
    cam.data.dof.focus_object = None

    def _dof_kf(focus_dist, fstop, frame):
        cam.data.dof.focus_distance = focus_dist
        cam.data.dof.aperture_fstop = fstop
        cam.data.dof.keyframe_insert(data_path="focus_distance", frame=frame)
        cam.data.dof.keyframe_insert(data_path="aperture_fstop",  frame=frame)

    def _frame_loc_rot(body_name, dist, h_off):
        bx, by, bz = LINEUP_POSITIONS.get(body_name, (0, 0, 0))
        loc   = (bx, by - dist, bz + h_off)
        pitch = math.degrees(math.atan2(h_off, dist))
        rot   = (math.radians(90 - pitch), 0, 0)
        return loc, rot

    def _kf(loc, rot, frame):
        cam.location       = loc
        cam.rotation_euler = rot
        cam.keyframe_insert(data_path="location",       frame=frame)
        cam.keyframe_insert(data_path="rotation_euler", frame=frame)

    sun_dist, sun_h_off = TOUR_CAMERA["Sun"]
    tight_loc, tight_rot = _frame_loc_rot("Sun", sun_dist * 0.35, sun_h_off * 0.6)
    _kf(tight_loc, tight_rot, frame=1)
    _dof_kf(sun_dist * 0.35, 2.0, frame=1)

    pulled_loc, pulled_rot = _frame_loc_rot("Sun", sun_dist, sun_h_off)
    _kf(pulled_loc, pulled_rot, frame=INTRO_END)
    _dof_kf(sun_dist, 2.8, frame=INTRO_END)

    HOLD_FRAMES = 40
    for body_idx, body_name in enumerate(TOUR_BODIES):
        tour_frame_start = TOUR_START + body_idx * TOUR_FRAMES_EACH
        hold_end         = tour_frame_start + HOLD_FRAMES
        dist, h_off      = TOUR_CAMERA.get(body_name, (8.0, 2.5))
        loc, rot         = _frame_loc_rot(body_name, dist, h_off)
        _kf(loc, rot, frame=tour_frame_start)
        _dof_kf(dist, 2.8, frame=tour_frame_start)
        _kf(loc, rot, frame=hold_end)
        _dof_kf(dist, 2.8, frame=hold_end)

    last_body          = TOUR_BODIES[-1]
    last_dist, last_h  = TOUR_CAMERA[last_body]
    last_loc, last_rot = _frame_loc_rot(last_body, last_dist, last_h)
    lineup_mid_x       = (LINEUP_POSITIONS["Sun"][0] + LINEUP_POSITIONS[last_body][0]) / 2.0

    _kf(last_loc, last_rot, frame=PULLBACK_START)
    _dof_kf(last_dist, 2.8, frame=PULLBACK_START)

    rise_start_loc = (lineup_mid_x, -40.0, 55.0)
    rise_start_rot = (math.radians(40), 0.0, 0.0)
    _kf(rise_start_loc, rise_start_rot, frame=PULLBACK_END)
    _dof_kf(60.0, 8.0, frame=PULLBACK_END)

    _kf(rise_start_loc, rise_start_rot, frame=SCATTER_START)
    _dof_kf(60.0, 8.0, frame=SCATTER_START)

    halfway_loc = (4.0, -18.0, 100.0)
    halfway_rot = (math.radians(15), 0.0, 0.0)
    _kf(halfway_loc, halfway_rot, frame=SCATTER_START + 90)
    _dof_kf(100.0, 11.0, frame=SCATTER_START + 90)

    zenith_loc = (2.0, -5.0, 130.0)
    zenith_rot = (math.radians(4), 0.0, 0.0)
    _kf(zenith_loc, zenith_rot, frame=SCATTER_END)
    _dof_kf(130.0, 16.0, frame=SCATTER_END)

    ease_down_loc = (0.0, -120.0, 60.0)
    ease_down_rot = (math.radians(75), 0.0, 0.0)
    _kf(ease_down_loc, ease_down_rot, frame=ORBIT_START + 80)
    _dof_kf(110.0, 14.0, frame=ORBIT_START + 80)

    final_loc = (0.0, -160.0, 45.0)
    final_rot = (math.radians(75), 0.0, 0.0)
    _kf(final_loc, final_rot, frame=ORBIT_START + 180)
    _dof_kf(160.0, 16.0, frame=ORBIT_START + 180)

    _kf(final_loc, final_rot, frame=TOTAL_FRAMES)
    _dof_kf(160.0, 16.0, frame=TOTAL_FRAMES)

    if cam.animation_data and cam.animation_data.action:
        for fc in cam.animation_data.action.fcurves:
            for kf in fc.keyframe_points:
                kf.interpolation     = "BEZIER"
                kf.handle_left_type  = "AUTO_CLAMPED"
                kf.handle_right_type = "AUTO_CLAMPED"

    if cam.data.animation_data and cam.data.animation_data.action:
        for fc in cam.data.animation_data.action.fcurves:
            if fc.data_path.startswith("dof."):
                for kf in fc.keyframe_points:
                    kf.interpolation     = "BEZIER"
                    kf.handle_left_type  = "AUTO_CLAMPED"
                    kf.handle_right_type = "AUTO_CLAMPED"

    lens_keys = [
        (1,                   85),
        (INTRO_END,           35),
        (TOUR_END,            35),
        (PULLBACK_END,        28),
        (SCATTER_START + 90,  20),
        (SCATTER_END,         18),
        (ORBIT_START + 180,   28),
        (TOTAL_FRAMES,        28),
    ]
    for frame, focal in lens_keys:
        cam.data.lens = focal
        cam.data.keyframe_insert(data_path="lens", frame=frame)

    print("  ✅ God's-Eye Overhead Sweep camera complete")
    return cam

def animate_planet_cinematic(name, orbit_r, period, anim_obj,
                              compressed_period, orbit_scale):
    lx, ly, lz = LINEUP_POSITIONS.get(name, (orbit_r, 0, 0))

    anim_obj.location = (lx, ly, lz)
    anim_obj.keyframe_insert(data_path="location", frame=1)
    anim_obj.keyframe_insert(data_path="location", frame=SCATTER_START - 1)

    start_angle   = math.radians(ORBIT_START_ANGLES.get(name, 0.0))
    orbit_start_x = math.cos(start_angle) * orbit_r
    orbit_start_y = math.sin(start_angle) * orbit_r

    mid_frame  = (SCATTER_START + SCATTER_END) // 2
    bow_factor = 0.25
    dx = orbit_start_x - lx
    dy = orbit_start_y - ly
    mid_x = (lx + orbit_start_x) * 0.5 + (-dy * bow_factor)
    mid_y = (ly + orbit_start_y) * 0.5 + ( dx * bow_factor)

    anim_obj.location = (lx, ly, 0.0)
    anim_obj.keyframe_insert(data_path="location", frame=SCATTER_START)

    anim_obj.location = (mid_x, mid_y, 0.0)
    anim_obj.keyframe_insert(data_path="location", frame=mid_frame)

    anim_obj.location = (orbit_start_x, orbit_start_y, 0.0)
    anim_obj.keyframe_insert(data_path="location", frame=SCATTER_END)

    if anim_obj.animation_data and anim_obj.animation_data.action:
        for fc in anim_obj.animation_data.action.fcurves:
            for kf in fc.keyframe_points:
                if kf.co[0] <= SCATTER_END:
                    kf.interpolation     = "BEZIER"
                    kf.handle_left_type  = "AUTO_CLAMPED"
                    kf.handle_right_type = "AUTO_CLAMPED"

    orbit_duration = compressed_period * orbit_scale
    steps          = 64
    num_keys       = int(ORBIT_FRAMES / orbit_duration * steps) + steps + 1

    for k in range(num_keys + 1):
        frame = ORBIT_START + k * (orbit_duration / steps)
        if frame > TOTAL_FRAMES:
            break
        angle = start_angle + (2 * math.pi * k) / steps
        anim_obj.location = (
            math.cos(angle) * orbit_r,
            math.sin(angle) * orbit_r,
            0.0,
        )
        anim_obj.keyframe_insert(data_path="location", frame=frame)

    if anim_obj.animation_data and anim_obj.animation_data.action:
        for fc in anim_obj.animation_data.action.fcurves:
            for kf in fc.keyframe_points:
                if kf.co[0] >= ORBIT_START:
                    kf.interpolation = "LINEAR"

def build_solar_system():
    clear_scene()
    setup_render_settings()

    print("\n📁 Checking textures...\n")
    for body, path in TEXTURE_URLS.items():
        status = "✅" if os.path.exists(path) else "❌ MISSING"
        print(f"  {status}  {body}: {path}")
    print()

    compressed_periods = {p[0]: math.log(p[3]) for p in PLANETS}
    max_compressed     = max(compressed_periods.values())
    ORBIT_FRAMES_SPEED_REF = 899
    orbit_scale = ORBIT_FRAMES_SPEED_REF / max_compressed

    print(f"  📐 Orbit scale: {orbit_scale:.2f} frames/log-unit  "
          f"(Pluto = {ORBIT_FRAMES} frames = 1 revolution)")
    for name, cp in compressed_periods.items():
        revs = ORBIT_FRAMES / (cp * orbit_scale)
        print(f"      {name:8s}: {cp * orbit_scale:6.1f} frames/rev  →  {revs:.2f} revolutions")
    print()

    # ── Sun ──────────────────────────────────────────────────────────────────
    sun_mat  = make_sun_material("Sun_Mat", TEXTURE_URLS.get("Sun"))
    sun_tilt = make_tilt_empty("Sun_TiltEmpty", AXIAL_TILTS.get("Sun", 0.0))

    lx, ly, lz = LINEUP_POSITIONS["Sun"]
    sun_tilt.location = (lx, ly, lz)

    sun_obj = add_sphere("Sun", SUN_RADIUS, (0, 0, 0), sun_mat)
    sun_obj.parent = sun_tilt

    animate_sun_lineup(sun_tilt)
    animate_pure_z_spin(sun_obj, SUN_ROTATION_PERIOD)

    add_starfield()

    # ── Orbit rings — created hidden, revealed at ORBIT_START ─────────────────
    for (name, size, orbit_r, period, fallback_color) in PLANETS:
        ring = add_orbit_ring(f"{name}_Orbit", orbit_r)
        ring.hide_render   = True
        ring.hide_viewport = True
        ring.keyframe_insert(data_path="hide_render",   frame=1)
        ring.keyframe_insert(data_path="hide_viewport", frame=1)
        ring.hide_render   = False
        ring.hide_viewport = False
        ring.keyframe_insert(data_path="hide_render",   frame=ORBIT_START)
        ring.keyframe_insert(data_path="hide_viewport", frame=ORBIT_START)

    # ── Planets ───────────────────────────────────────────────────────────────
    planet_objects    = {}
    earth_tilt_empty  = None
    earth_orbit_pivot = None

    for (name, size, orbit_r, period, fallback_color) in PLANETS:
        tex_path       = TEXTURE_URLS.get(name)
        mat            = make_planet_material(name, tex_path, fallback_color)
        tilt_deg       = AXIAL_TILTS.get(name, 0.0)
        base_direction = ROTATION_DIRECTION.get(name, 1)
        direction      = world_spin_direction(tilt_deg, base_direction)
        cp             = compressed_periods[name]

        if name == "Saturn":
            bpy.ops.object.empty_add(type="PLAIN_AXES", location=(0, 0, 0))
            orbit_pivot = bpy.context.active_object
            orbit_pivot.name               = "Saturn_OrbitPivot"
            orbit_pivot.empty_display_size = 0.01

            tilt_empty = make_tilt_empty("Saturn_TiltEmpty", tilt_deg, parent=orbit_pivot)
            planet     = add_sphere(name, size, (0, 0, 0), mat)
            planet.parent = tilt_empty

            create_saturn_rings(orbit_pivot, planet_radius=size,
                                texture_path=TEXTURE_URLS.get("Saturn_Ring"))

            animate_pure_z_spin(planet, ROTATION_PERIODS.get(name, 60), direction=direction)
            animate_planet_cinematic(name, orbit_r, period,
                                     orbit_pivot, cp, orbit_scale)

        else:
            tilt_empty = make_tilt_empty(f"{name}_TiltEmpty", tilt_deg)
            planet     = add_sphere(name, size, (0, 0, 0), mat)
            planet.parent = tilt_empty

            animate_pure_z_spin(planet, ROTATION_PERIODS.get(name, 60), direction=direction)
            animate_planet_cinematic(name, orbit_r, period,
                                     tilt_empty, cp, orbit_scale)

            if name == "Earth":
                earth_tilt_empty  = tilt_empty
                earth_orbit_pivot = bpy.data.objects.new("Earth_OrbitPivot", None)
                bpy.context.collection.objects.link(earth_orbit_pivot)
                earth_orbit_pivot.empty_display_type = "PLAIN_AXES"
                earth_orbit_pivot.empty_display_size = 0.01
                copy_loc = earth_orbit_pivot.constraints.new("COPY_LOCATION")
                copy_loc.target = earth_tilt_empty

        planet_objects[name] = planet

    # ── Moon ─────────────────────────────────────────────────────────────────
    if earth_orbit_pivot is not None:
        build_moon(earth_orbit_pivot)
    else:
        print("  ⚠️ Earth orbit pivot not found — Moon skipped.")

    # ── Hubble ────────────────────────────────────────────────────────────────
    if earth_orbit_pivot is not None:
        build_hubble(earth_orbit_pivot)
    else:
        print("  ⚠️ Earth orbit pivot not found — Hubble skipped.")

    # ── Asteroid Belt ─────────────────────────────────────────────────────────
    build_asteroid_belt(
        num_asteroids=300,
        inner_r=29.0,
        outer_r=33.0,
        max_z=0.6,
        seed=42,
        num_rogues=20
    )

    # ── Kuiper Belt ───────────────────────────────────────────────────────────
    build_kuiper_belt(
        num_kbos=500,
        inner_r=78.0,
        outer_r=98.0,
        max_z=3.0,
        seed=99,
        num_rogues=35
    )

    # ── Cinematic camera ──────────────────────────────────────────────────────
    setup_cinematic_camera(planet_objects)

    setup_lighting()
    force_rendered_viewport()

    for obj in bpy.data.objects:
        obj.display_type = 'TEXTURED'

    # ── Hide local orbit rings (Moon, HST) until orbital phase ────────────────
    for obj in bpy.data.objects:
        if "OrbitRing" in obj.name:
            obj.hide_render   = True
            obj.hide_viewport = True
            obj.keyframe_insert(data_path="hide_render",   frame=1)
            obj.keyframe_insert(data_path="hide_viewport", frame=1)
            obj.hide_render   = False
            obj.hide_viewport = False
            obj.keyframe_insert(data_path="hide_render",   frame=ORBIT_START)
            obj.keyframe_insert(data_path="hide_viewport", frame=ORBIT_START)

    # ── Hide all empties from viewport and render ─────────────────────────────
    for obj in bpy.data.objects:
        if obj.type == 'EMPTY':
            obj.hide_set(True)
            obj.hide_render = True

    bpy.context.scene.frame_set(1)

    print("\n✅  Done!")
    print(f"    🎬 Timeline: {TOTAL_FRAMES} frames total")
    print(f"    🌍 Orbital phase: frames {ORBIT_START}–{TOTAL_FRAMES} ({ORBIT_FRAMES} frames)")
    print(f"    🔁 Every planet completes ≥ 1 full revolution (Pluto = exactly 1)")
    print(f"    ⚡ Inner planets orbit much faster (Mercury ≈ {ORBIT_FRAMES / (compressed_periods['Mercury'] * orbit_scale):.1f}× revolutions)")


build_solar_system()

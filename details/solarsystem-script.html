def clear_scene():
    bpy.ops.object.select_all(action="SELECT")
    bpy.ops.object.delete(use_global=False)
    for mesh in list(bpy.data.meshes):
        bpy.data.meshes.remove(mesh)
    for mat in list(bpy.data.materials):
        bpy.data.materials.remove(mat)

def _tex_uv_chain(nt, texture_path):
    tex_coord = nt.nodes.new("ShaderNodeTexCoord"); tex_coord.location = (-600, 100)
    mapping   = nt.nodes.new("ShaderNodeMapping");   mapping.location   = (-400, 100)
    img_tex   = nt.nodes.new("ShaderNodeTexImage");  img_tex.location   = (-150, 100)
    nt.links.new(tex_coord.outputs["UV"], mapping.inputs["Vector"])
    nt.links.new(mapping.outputs["Vector"], img_tex.inputs["Vector"])
    if texture_path and os.path.exists(texture_path):
        try:
            img_tex.image = bpy.data.images.load(texture_path, check_existing=True)
            return img_tex, True
        except Exception as e:
            print(f"    ⚠️  Texture load failed: {e}")
    return img_tex, False


def _uv_noise(nt, scale, detail, roughness, distortion, offset_x=0, offset_y=0):
    tc = nt.nodes.new("ShaderNodeTexCoord"); tc.location = (offset_x - 400, offset_y)
    mp = nt.nodes.new("ShaderNodeMapping");   mp.location = (offset_x - 200, offset_y)
    ns = nt.nodes.new("ShaderNodeTexNoise");  ns.location = (offset_x,       offset_y)
    ns.inputs["Scale"].default_value      = scale
    ns.inputs["Detail"].default_value     = detail
    ns.inputs["Roughness"].default_value  = roughness
    ns.inputs["Distortion"].default_value = distortion
    nt.links.new(tc.outputs["UV"], mp.inputs["Vector"])
    nt.links.new(mp.outputs["Vector"], ns.inputs["Vector"])
    return ns

def make_sun_material(name, texture_path):
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    nt = mat.node_tree
    nt.nodes.clear()

    output   = nt.nodes.new("ShaderNodeOutputMaterial"); output.location   = (1200, 0)
    emission = nt.nodes.new("ShaderNodeEmission");       emission.location  = (900,  0)
    emission.inputs["Strength"].default_value = 2.0

    img_tex, loaded = _tex_uv_chain(nt, texture_path)

    if loaded:
        ns = _uv_noise(nt, 120, 14, 0.8, 0.3, offset_x=-600, offset_y=-200)

        nt.links.new(ns.outputs["Color"],      mix_rgb.inputs["Color2"])
        nt.links.new(mix_rgb.outputs["Color"], emission.inputs["Color"])
        print("  ✅ Sun texture loaded + granulation overlay")
    else:
        print("  ⚠️  Using procedural sun (granulation + limb darkening)")
        ns1 = _uv_noise(nt, 8, 14, 0.72, 0.6, offset_x=-600, offset_y=0)
        ramp1 = nt.nodes.new("ShaderNodeValToRGB"); ramp1.location = (-100, 0)

        nt.links.new(mult.outputs["Color"],  emission.inputs["Color"])

    nt.links.new(emission.outputs["Emission"], output.inputs["Surface"])
    return mat

PLANET_PRESETS = {
    "Mercury": (0.95, 0.05, 0.50, False),
    "Venus":   (0.65, 0.00, 1.00, True),
    "Earth":   (0.70, 0.00, 0.70, True),
    "Mars":    (0.90, 0.05, 0.55, False),

    "Uranus":  (0.45, 0.00, 1.30, True),
    "Neptune": (0.42, 0.00, 1.40, True),

def make_planet_material(name, texture_path, base_color):
    roughness, metallic, emission_strength, has_atmo = PLANET_PRESETS.get(
        name, (0.80, 0.00, 0.80, False))

    mat_name = name + "_Mat" if not name.endswith("_Mat") else name
    mat = bpy.data.materials.new(name=mat_name)
    mat.use_nodes = True
    nt = mat.node_tree
    nt.nodes.clear()

    output   = nt.nodes.new("ShaderNodeOutputMaterial"); output.location   = (1000, 0)
    add_sh   = nt.nodes.new("ShaderNodeAddShader");      add_sh.location   = (780,  0)
    bsdf     = nt.nodes.new("ShaderNodeBsdfPrincipled"); bsdf.location     = (480,  80)
    bsdf.inputs["Roughness"].default_value = roughness
    bsdf.inputs["Metallic"].default_value  = metallic
    emission = nt.nodes.new("ShaderNodeEmission");       emission.location = (480, -120)
    emission.inputs["Strength"].default_value = emission_strength

    img_tex, loaded = _tex_uv_chain(nt, texture_path)

    if loaded:
        color_source = img_tex.outputs["Color"]
    else:
        color_source = _make_procedural_planet(nt, name, base_color)

    nt.links.new(color_source,              bsdf.inputs["Base Color"])
    nt.links.new(color_source,              emission.inputs["Color"])
    nt.links.new(bsdf.outputs["BSDF"],      add_sh.inputs[0])
    nt.links.new(emission.outputs["Emission"], add_sh.inputs[1])

    if has_atmo:
        atmo_emission = _add_atmosphere_rim(nt, name)
        add_sh2 = nt.nodes.new("ShaderNodeAddShader"); add_sh2.location = (900, 0)
        nt.links.new(add_sh.outputs["Shader"],         add_sh2.inputs[0])
        nt.links.new(atmo_emission.outputs["Emission"], add_sh2.inputs[1])
        nt.links.new(add_sh2.outputs["Shader"],        output.inputs["Surface"])
    else:
        nt.links.new(add_sh.outputs["Shader"], output.inputs["Surface"])

    return mat

def _add_atmosphere_rim(nt, name):
    atmo_colors = {
        "Earth":   (0.35, 0.60, 1.00),
        "Venus":   (0.90, 0.75, 0.35),
        "Uranus":  (0.45, 0.85, 0.90),
        "Neptune": (0.25, 0.50, 1.00),
    }
    atmo_col = atmo_colors.get(name, (0.6, 0.7, 1.0))
    fresnel = nt.nodes.new("ShaderNodeFresnel");  fresnel.location  = (250, -350)
    fresnel.inputs["IOR"].default_value = 1.05
    ramp    = nt.nodes.new("ShaderNodeValToRGB"); ramp.location      = (450, -350)
    ramp.color_ramp.elements[0].position = 0.55
    ramp.color_ramp.elements[0].color    = (0.0, 0.0, 0.0, 1.0)
    ramp.color_ramp.elements[1].position = 0.88
    ramp.color_ramp.elements[1].color    = (*atmo_col, 1.0)
    emission = nt.nodes.new("ShaderNodeEmission"); emission.location = (680, -350)
    emission.inputs["Color"].default_value    = (*atmo_col, 1.0)
    emission.inputs["Strength"].default_value = 0.6
    nt.links.new(fresnel.outputs["Fac"], ramp.inputs["Fac"])
    nt.links.new(ramp.outputs["Color"],  emission.inputs["Color"])
    return emission

def make_ring_material(img, alpha_fallback=0.75, ring_name=""):
    mat = bpy.data.materials.new("RingMat_" + ring_name)
    mat.use_nodes = True
    mat.use_backface_culling = False
    if hasattr(mat, "blend_method"):         mat.blend_method = "HASHED"
    if hasattr(mat, "shadow_method"):        mat.shadow_method = "HASHED"
    if hasattr(mat, "use_transparent_shadow"): mat.use_transparent_shadow = True

    nt = mat.node_tree; nt.nodes.clear()

    out      = nt.nodes.new("ShaderNodeOutputMaterial")
    mix      = nt.nodes.new("ShaderNodeMixShader")
    transp   = nt.nodes.new("ShaderNodeBsdfTransparent")
    add_sh   = nt.nodes.new("ShaderNodeAddShader")
    bsdf     = nt.nodes.new("ShaderNodeBsdfPrincipled")
    bsdf.inputs["Roughness"].default_value = 0.95
    bsdf.inputs["Metallic"].default_value  = 0.10
    emission = nt.nodes.new("ShaderNodeEmission")
    emission.inputs["Strength"].default_value = 0.35
    coord    = nt.nodes.new("ShaderNodeTexCoord")
    mapping  = nt.nodes.new("ShaderNodeMapping")
    mapping.inputs["Location"].default_value = (0.0, 0.0, 0.0)
    mapping.inputs["Scale"].default_value    = (1.0, 1.0, 1.0)
    nt.links.new(coord.outputs["UV"], mapping.inputs["Vector"])

    if img:
        tex = nt.nodes.new("ShaderNodeTexImage")
        tex.image     = img
        tex.extension = "EXTEND"
        nt.links.new(mapping.outputs["Vector"], tex.inputs["Vector"])
        bc = nt.nodes.new("ShaderNodeBrightContrast")
        bc.inputs["Bright"].default_value   = 0.08
        bc.inputs["Contrast"].default_value = 0.40
        nt.links.new(tex.outputs["Color"], bc.inputs["Color"])
        nt.links.new(bc.outputs["Color"],  bsdf.inputs["Base Color"])
        nt.links.new(bc.outputs["Color"],  emission.inputs["Color"])
        sep_uv = nt.nodes.new("ShaderNodeSeparateXYZ")
        nt.links.new(mapping.outputs["Vector"], sep_uv.inputs["Vector"])
        gradient_ramp = nt.nodes.new("ShaderNodeValToRGB")
        gradient_ramp.color_ramp.interpolation = "EASE"
        gradient_ramp.color_ramp.elements[0].position = 0.0
        gradient_ramp.color_ramp.elements[0].color    = (0.0, 0.0, 0.0, 1.0)
        gradient_ramp.color_ramp.elements[1].position = 1.0
        gradient_ramp.color_ramp.elements[1].color    = (1.0, 1.0, 1.0, 1.0)
        inner_edge = gradient_ramp.color_ramp.elements.new(0.06)
        inner_edge.color = (1.0, 1.0, 1.0, 1.0)
        outer_edge = gradient_ramp.color_ramp.elements.new(0.93)
        outer_edge.color = (1.0, 1.0, 1.0, 1.0)
        nt.links.new(sep_uv.outputs["X"], gradient_ramp.inputs["Fac"])
        alpha_math = nt.nodes.new("ShaderNodeMath")
        alpha_math.operation = "MULTIPLY"
        nt.links.new(tex.outputs["Alpha"],           alpha_math.inputs[0])
        nt.links.new(gradient_ramp.outputs["Color"], alpha_math.inputs[1])
        nt.links.new(alpha_math.outputs["Value"], mix.inputs["Fac"])
    else:
        ns_r = nt.nodes.new("ShaderNodeTexNoise")
        ns_r.inputs["Scale"].default_value     = 40.0
        ns_r.inputs["Detail"].default_value    = 8.0
        ns_r.inputs["Roughness"].default_value = 0.70
        nt.links.new(mapping.outputs["Vector"], ns_r.inputs["Vector"])
        ramp_col = nt.nodes.new("ShaderNodeValToRGB")
        ramp_col.color_ramp.elements[0].color = (0.50, 0.45, 0.36, 1.0)
        ramp_col.color_ramp.elements[1].color = (0.80, 0.74, 0.60, 1.0)
        nt.links.new(ns_r.outputs["Fac"],       ramp_col.inputs["Fac"])
        nt.links.new(ramp_col.outputs["Color"], bsdf.inputs["Base Color"])
        nt.links.new(ramp_col.outputs["Color"], emission.inputs["Color"])
        sep_uv = nt.nodes.new("ShaderNodeSeparateXYZ")
        nt.links.new(mapping.outputs["Vector"], sep_uv.inputs["Vector"])
        gradient_ramp = nt.nodes.new("ShaderNodeValToRGB")
        gradient_ramp.color_ramp.interpolation = "EASE"
        gradient_ramp.color_ramp.elements[0].position = 0.0
        gradient_ramp.color_ramp.elements[0].color    = (0.0, 0.0, 0.0, 1.0)
        gradient_ramp.color_ramp.elements[1].position = 1.0
        gradient_ramp.color_ramp.elements[1].color    = (1.0, 1.0, 1.0, 1.0)
        inner_edge = gradient_ramp.color_ramp.elements.new(0.06)
        inner_edge.color = (1.0, 1.0, 1.0, 1.0)
        outer_edge = gradient_ramp.color_ramp.elements.new(0.93)
        outer_edge.color = (1.0, 1.0, 1.0, 1.0)
        nt.links.new(sep_uv.outputs["X"], gradient_ramp.inputs["Fac"])
        alpha_math = nt.nodes.new("ShaderNodeMath")
        alpha_math.operation = "MULTIPLY"
        alpha_math.inputs[1].default_value = alpha_fallback
        nt.links.new(gradient_ramp.outputs["Color"], alpha_math.inputs[0])
        nt.links.new(alpha_math.outputs["Value"], mix.inputs["Fac"])

    nt.links.new(bsdf.outputs["BSDF"],         add_sh.inputs[0])
    nt.links.new(emission.outputs["Emission"], add_sh.inputs[1])
    nt.links.new(transp.outputs["BSDF"],       mix.inputs[1])
    nt.links.new(add_sh.outputs["Shader"],     mix.inputs[2])
    nt.links.new(mix.outputs["Shader"],        out.inputs["Surface"])
    return mat

    ring_specs = [
        (3.50 * scale, 4.00 * scale, "D_Ring", 0.00, 0.08, 0.30),
        (4.05 * scale, 5.20 * scale, "C_Ring", 0.08, 0.30, 0.55),
        (5.25 * scale, 7.40 * scale, "B_Ring", 0.30, 0.68, 0.95),
        (7.60 * scale, 9.50 * scale, "A_Ring", 0.68, 0.95, 0.90),
        (9.70 * scale, 9.90 * scale, "F_Ring", 0.95, 1.00, 0.45),
    ]

    ring_img = None
    if texture_path and os.path.exists(texture_path):
        ring_img = bpy.data.images.get("Saturn_Rings") or bpy.data.images.load(texture_path)
        ring_img.name = "Saturn_Rings"
        ring_img.colorspace_settings.name = "sRGB"

    ring_parent = bpy.data.objects.new("Saturn_RingSystem", None)
    bpy.context.collection.objects.link(ring_parent)
    ring_parent.empty_display_type = "PLAIN_AXES"
    ring_parent.empty_display_size = 0.01
    ring_parent.rotation_euler = (math.radians(26.7), 0.0, 0.0)
    ring_parent.parent = parent_obj

    SEGMENTS = 256
    for inner, outer, rname, u0, u1, alpha in ring_specs:
        bm = bmesh.new()
        uv_layer = bm.loops.layers.uv.new("UVMap")
        cos_a = [math.cos(2.0 * math.pi * i / SEGMENTS) for i in range(SEGMENTS)]
        sin_a = [math.sin(2.0 * math.pi * i / SEGMENTS) for i in range(SEGMENTS)]
        for i in range(SEGMENTS):
            j = (i + 1) % SEGMENTS
            vi0 = bm.verts.new((inner * cos_a[i], inner * sin_a[i], 0.0))
            vo0 = bm.verts.new((outer * cos_a[i], outer * sin_a[i], 0.0))
            vo1 = bm.verts.new((outer * cos_a[j], outer * sin_a[j], 0.0))
            vi1 = bm.verts.new((inner * cos_a[j], inner * sin_a[j], 0.0))
            face = bm.faces.new([vi0, vo0, vo1, vi1])
            t0 = i / SEGMENTS; t1 = (i + 1) / SEGMENTS
            face.loops[0][uv_layer].uv = (u0, t0)
            face.loops[1][uv_layer].uv = (u1, t0)
            face.loops[2][uv_layer].uv = (u1, t1)
            face.loops[3][uv_layer].uv = (u0, t1)
        mesh = bpy.data.meshes.new(rname + "_Mesh")
        bm.to_mesh(mesh); bm.free()
        mesh.uv_layers[0].name = "UVMap"
        ring = bpy.data.objects.new(rname, mesh)
        bpy.context.collection.objects.link(ring)
        ring.data.materials.append(
            make_ring_material(ring_img, alpha_fallback=alpha, ring_name=rname))
        bpy.context.view_layer.objects.active = ring
        ring.select_set(True)
        bpy.ops.object.shade_flat()
        ring.parent = ring_parent

    print("  ✅ Saturn rings built with 26.7° tilt")
    return ring_parent

def _spawn_body(parent_empty, mat, x, y, z, size_ranges, subdivisions=1):
    size = random.choice(size_ranges)
    bpy.ops.mesh.primitive_ico_sphere_add(
        radius=size, subdivisions=subdivisions, location=(x, y, z))
    obj = bpy.context.active_object
    obj.rotation_euler = (
        random.uniform(0, math.pi * 2),
        random.uniform(0, math.pi * 2),
        random.uniform(0, math.pi * 2),
    )
    sx = random.uniform(0.6, 1.4)
    sy = random.uniform(0.6, 1.4)
    sz = random.uniform(0.5, 1.2)
    obj.scale = (sx, sy, sz)
    bpy.ops.object.transform_apply(scale=True)
    obj.data.materials.append(mat)
    obj.parent = parent_empty
    return obj

def build_asteroid_belt(num_asteroids=500, inner_r=26.0, outer_r=31.5,
                        max_z=0.6, seed=42, num_rogues=20):
    print(f"\n  🪨 Building asteroid belt ({num_asteroids} rocks + {num_rogues} rogues)...")
    random.seed(seed)

    mat = make_asteroid_material()

    belt_empty = bpy.data.objects.new("AsteroidBelt", None)
    belt_empty.empty_display_type = "PLAIN_AXES"
    belt_empty.empty_display_size = 0.01
    bpy.context.collection.objects.link(belt_empty)

    mid_r = (inner_r + outer_r) / 2.0

    main_sizes = (
        [random.uniform(0.04, 0.09) for _ in range(40)] +
        [random.uniform(0.09, 0.16) for _ in range(25)] +
        [random.uniform(0.16, 0.26) for _ in range(8)]  +
        [random.uniform(0.26, 0.50) for _ in range(2)]
    )

    for i in range(num_asteroids):
        radius = random.triangular(inner_r, outer_r, mid_r)
        angle  = random.uniform(0.0, 2.0 * math.pi)
        x = math.cos(angle) * radius
        y = math.sin(angle) * radius
        z = random.gauss(0.0, max_z / 2.5)
        z = max(-max_z, min(max_z, z))
        obj = _spawn_body(belt_empty, mat, x, y, z, main_sizes)
        obj.name = f"Asteroid_{i:03d}"

    rogue_sizes = [random.uniform(0.05, 0.20) for _ in range(20)]
    rogue_span  = (outer_r - inner_r)

    for i in range(num_rogues):
        if random.random() < 0.5:
            radius = random.uniform(inner_r - rogue_span * 0.6, inner_r - 0.5)
        else:
            radius = random.uniform(outer_r + 0.5, outer_r + rogue_span * 0.6)
        radius = max(radius, 1.0)
        angle  = random.uniform(0.0, 2.0 * math.pi)
        x = math.cos(angle) * radius
        y = math.sin(angle) * radius
        z = random.gauss(0.0, max_z * 1.8)
        z = max(-max_z * 4.0, min(max_z * 4.0, z))
        obj = _spawn_body(belt_empty, mat, x, y, z, rogue_sizes)
        obj.name = f"Asteroid_Rogue_{i:02d}"

    bpy.ops.mesh.primitive_torus_add(
        major_radius=mid_r, minor_radius=0.012,
        major_segments=128, minor_segments=6, location=(0, 0, 0))
    guide = bpy.context.active_object
    guide.name = "AsteroidBelt_OrbitGuide"
    guide_mat = bpy.data.materials.new("AsteroidBelt_Guide_Mat")
    guide_mat.use_nodes = True
    gnt = guide_mat.node_tree; gnt.nodes.clear()
    ge = gnt.nodes.new("ShaderNodeEmission")
    ge.inputs["Color"].default_value    = (0.50, 0.45, 0.38, 1.0)
    ge.inputs["Strength"].default_value = 0.05
    go = gnt.nodes.new("ShaderNodeOutputMaterial")
    gnt.links.new(ge.outputs["Emission"], go.inputs["Surface"])
    guide.data.materials.append(guide_mat)
    guide.parent = belt_empty

    print(f"  ✅ Asteroid belt — {num_asteroids} rocks (r={inner_r}-{outer_r}) + {num_rogues} rogues")
    return belt_empty

def build_kuiper_belt(num_kbos=700, inner_r=68.0, outer_r=88.0,
                      max_z=3.0, seed=99, num_rogues=35):
    print(f"\n  ❄️  Building Kuiper Belt ({num_kbos} KBOs + {num_rogues} scattered-disc objects)...")
    random.seed(seed)

    mat = make_kuiper_material()

    belt_empty = bpy.data.objects.new("KuiperBelt", None)
    belt_empty.empty_display_type = "PLAIN_AXES"
    belt_empty.empty_display_size = 0.01
    bpy.context.collection.objects.link(belt_empty)

    for obj in bpy.data.objects:
        if obj.type == "CAMERA":
            obj.data.clip_end = 1000.0

    mid_r        = (inner_r + outer_r) / 2.0
    inner_edge   = inner_r
    classic_edge = inner_r + (outer_r - inner_r) * 0.55
    outer_edge   = outer_r

    main_sizes = (
        [random.uniform(0.12, 0.20) for _ in range(38)] +
        [random.uniform(0.20, 0.32) for _ in range(28)] +
        [random.uniform(0.32, 0.50) for _ in range(18)] +
        [random.uniform(0.50, 0.70) for _ in range(10)] +
        [random.uniform(0.70, 0.85) for _ in range(4)]
    )

    inner_count = int(num_kbos * 0.65)
    outer_count = num_kbos - inner_count

    for i in range(inner_count):
        radius = random.triangular(inner_edge, classic_edge, (inner_edge + classic_edge) / 2.0)
        angle  = random.uniform(0.0, 2.0 * math.pi)
        x = math.cos(angle) * radius
        y = math.sin(angle) * radius
        z = random.gauss(0.0, max_z / 2.0)
        z = max(-max_z, min(max_z, z))
        obj = _spawn_body(belt_empty, mat, x, y, z, main_sizes)
        obj.name = f"KBO_{i:03d}"

    for i in range(outer_count):
        radius = random.triangular(classic_edge, outer_edge, (classic_edge + outer_edge) / 2.0)
        angle  = random.uniform(0.0, 2.0 * math.pi)
        x = math.cos(angle) * radius
        y = math.sin(angle) * radius
        z = random.gauss(0.0, max_z / 1.6)
        z = max(-max_z * 1.4, min(max_z * 1.4, z))
        obj = _spawn_body(belt_empty, mat, x, y, z, main_sizes)
        obj.name = f"KBO_{inner_count + i:03d}"

    rogue_span  = (outer_r - inner_r)
    rogue_sizes = (
        [random.uniform(0.15, 0.35) for _ in range(18)] +
        [random.uniform(0.35, 0.65) for _ in range(12)] +
        [random.uniform(0.65, 0.90) for _ in range(5)]
    )

    for i in range(num_rogues):
        if random.random() < 0.20:
            radius = random.uniform(inner_r - rogue_span * 0.25, inner_r - 0.5)
        else:
            radius = random.uniform(outer_r + 0.5, outer_r + rogue_span * 1.0)
        radius = max(radius, inner_r * 0.8)

        angle  = random.uniform(0.0, 2.0 * math.pi)
        x = math.cos(angle) * radius
        y = math.sin(angle) * radius
        z = random.gauss(0.0, max_z * 2.2)
        z = max(-max_z * 6.0, min(max_z * 6.0, z))

        obj = _spawn_body(belt_empty, mat, x, y, z, rogue_sizes)
        obj.name = f"KBO_Scattered_{i:02d}"

    print(f"  ✅ Kuiper Belt — {num_kbos} KBOs (r={inner_r}-{outer_r}) + {num_rogues} scattered-disc objects")
    return belt_empty

def make_hubble_solar_panel_material():
    """

    """
    mat = bpy.data.materials.new(name="Hubble_SolarPanel_Mat")
    mat.use_nodes = True
    nt = mat.node_tree
    nt.nodes.clear()

    output   = nt.nodes.new("ShaderNodeOutputMaterial"); output.location = (900, 0)
    add_sh   = nt.nodes.new("ShaderNodeAddShader");      add_sh.location = (700, 0)

    bsdf.inputs["Metallic"].default_value  = 0.10
    emission.inputs["Strength"].default_value = 0.80

    tex_coord = nt.nodes.new("ShaderNodeTexCoord"); tex_coord.location = (-700, 0)
    mapping   = nt.nodes.new("ShaderNodeMapping");   mapping.location  = (-500, 0)
    # Scale U heavily to create many narrow cell columns
    mapping.inputs["Scale"].default_value = (18.0, 6.0, 1.0)

    # Wave texture for cell columns (U direction)
    wave_u = nt.nodes.new("ShaderNodeTexWave"); wave_u.location = (-280, 100)
    wave_u.wave_type    = "BANDS"
    wave_u.bands_direction = "X"
    wave_u.inputs["Scale"].default_value      = 1.0
    wave_u.inputs["Distortion"].default_value = 0.0
    wave_u.inputs["Detail"].default_value     = 0.0

    # Wave texture for cell rows (V direction)
    wave_v = nt.nodes.new("ShaderNodeTexWave"); wave_v.location = (-280, -120)
    wave_v.wave_type    = "BANDS"
    wave_v.bands_direction = "Y"
    wave_v.inputs["Scale"].default_value      = 1.0
    wave_v.inputs["Distortion"].default_value = 0.0
    wave_v.inputs["Detail"].default_value     = 0.0

    nt.links.new(tex_coord.outputs["UV"], mapping.inputs["Vector"])
    nt.links.new(mapping.outputs["Vector"], wave_u.inputs["Vector"])
    nt.links.new(mapping.outputs["Vector"], wave_v.inputs["Vector"])

    # Threshold each wave to a thin grid line
    ramp_u = nt.nodes.new("ShaderNodeValToRGB"); ramp_u.location = (-60, 100)
    ramp_u.color_ramp.interpolation = "CONSTANT"
    ramp_u.color_ramp.elements[0].position = 0.44
    ramp_u.color_ramp.elements[0].color    = (0.04, 0.08, 0.28, 1.0)  # dark cell
    ramp_u.color_ramp.elements[1].position = 0.56
    ramp_u.color_ramp.elements[1].color    = (0.70, 0.76, 0.90, 1.0)  # bright grid line
    nt.links.new(wave_u.outputs["Fac"], ramp_u.inputs["Fac"])

    ramp_v = nt.nodes.new("ShaderNodeValToRGB"); ramp_v.location = (-60, -120)
    ramp_v.color_ramp.interpolation = "CONSTANT"
    ramp_v.color_ramp.elements[0].position = 0.44
    ramp_v.color_ramp.elements[0].color    = (0.04, 0.08, 0.28, 1.0)
    ramp_v.color_ramp.elements[1].position = 0.56
    ramp_v.color_ramp.elements[1].color    = (0.70, 0.76, 0.90, 1.0)
    nt.links.new(wave_v.outputs["Fac"], ramp_v.inputs["Fac"])

    # Combine both grid directions: add them, then clamp
    add_grid = nt.nodes.new("ShaderNodeMixRGB"); add_grid.location = (180, 0)
    add_grid.blend_type = "LIGHTEN"
    add_grid.inputs["Fac"].default_value = 1.0
    nt.links.new(ramp_u.outputs["Color"], add_grid.inputs["Color1"])
    nt.links.new(ramp_v.outputs["Color"], add_grid.inputs["Color2"])

    # Base deep-blue cell colour underneath the grid
    cell_col = nt.nodes.new("ShaderNodeRGB"); cell_col.location = (180, -250)
    cell_col.outputs[0].default_value = (0.03, 0.06, 0.22, 1.0)

    # Mix: grid lines lighten the base cell colour
    final_mix = nt.nodes.new("ShaderNodeMixRGB"); final_mix.location = (380, -100)
    final_mix.blend_type = "ADD"
    final_mix.inputs["Fac"].default_value = 0.18
    nt.links.new(cell_col.outputs["Color"],  final_mix.inputs["Color1"])
    nt.links.new(add_grid.outputs["Color"],  final_mix.inputs["Color2"])

    # Emissive version — faint blue glow from the panel face
    emit_col = nt.nodes.new("ShaderNodeRGB"); emit_col.location = (380, -300)
    emit_col.outputs[0].default_value = (0.06, 0.14, 0.55, 1.0)

    nt.links.new(final_mix.outputs["Color"], bsdf.inputs["Base Color"])
    nt.links.new(emit_col.outputs["Color"],  emission.inputs["Color"])
    nt.links.new(bsdf.outputs["BSDF"],         add_sh.inputs[0])
    nt.links.new(emission.outputs["Emission"], add_sh.inputs[1])
    nt.links.new(add_sh.outputs["Shader"],     output.inputs["Surface"])
    return mat

def build_hubble(earth_tilt_empty):
    """
    Construct a recognisable low-poly Hubble Space Telescope and place it
    on a close, fast, inclined orbit around Earth.

    Shape summary (all sizes relative to HST_SIZE):
      • Main tube  — long cylinder (the optical tube assembly)
      • Front baffle cap — flat disc closing the aperture end
      • Aft shroud  — slightly wider short cylinder at the back
      • Two solar array wings — large flat rectangular panels
      • Two high-gain antenna dishes — small cylinders + discs
      • Four fine-guidance sensor bays — small boxes on the tube sides
    """
    print("\n  🔭 Building Hubble Space Telescope...")

    HST_SIZE            = 0.06   # global scale factor
    ORBIT_RADIUS        = 1.6
    ORBIT_PERIOD        = 4
    INCLINATION_DEG     = 28.5   # Hubble's real orbital inclination

    body_mat    = make_hubble_body_material()
    panel_mat   = make_hubble_solar_panel_material()
    aperture_mat = make_hubble_aperture_material()

    # ── Orbital hierarchy ────────────────────────────────────────────────────
    incl_empty = bpy.data.objects.new("HST_InclinationEmpty", None)
    bpy.context.collection.objects.link(incl_empty)
    incl_empty.empty_display_type = "PLAIN_AXES"
    incl_empty.empty_display_size = 0.01
    incl_empty.rotation_mode  = "XYZ"
    incl_empty.rotation_euler = (math.radians(INCLINATION_DEG), 0.0, 0.0)
    incl_empty.parent = earth_tilt_empty

    orbit_empty = bpy.data.objects.new("HST_OrbitEmpty", None)
    bpy.context.collection.objects.link(orbit_empty)
    orbit_empty.empty_display_type = "PLAIN_AXES"
    orbit_empty.empty_display_size = 0.01
    orbit_empty.parent = incl_empty

    # Thin orbit guide ring
    add_local_orbit_ring("HST_OrbitRing", ORBIT_RADIUS, incl_empty)

    # ── Root empty — all HST parts parented here ──────────────────────────
    hst_root = bpy.data.objects.new("HST_Root", None)
    bpy.context.collection.objects.link(hst_root)
    hst_root.empty_display_type = "PLAIN_AXES"
    hst_root.empty_display_size = 0.01
    hst_root.parent = orbit_empty

    # ── Helper: add a named mesh part ────────────────────────────────────────
    def _add_part(obj, name, mat):
        obj.name      = name
        obj.data.name = name + "_Mesh"
        obj.data.materials.append(mat)
        obj.parent = hst_root
        return obj

    # ── Main optical tube (cylinder along local X axis) ──────────────────────
    # Blender cylinders are along Z; we rotate to align with X.
    tube_length = HST_SIZE * 7.0
    tube_radius = HST_SIZE * 0.85
    bpy.ops.mesh.primitive_cylinder_add(
        radius=tube_radius, depth=tube_length,
        vertices=32, location=(0, 0, 0))
    tube = bpy.context.active_object
    tube.rotation_euler = (0.0, math.radians(90), 0.0)
    bpy.ops.object.transform_apply(rotation=True)
    _add_part(tube, "HST_Tube", body_mat)
    bpy.ops.object.shade_smooth()

    # ── Front aperture cap (dark disc at +X end) ─────────────────────────────
    bpy.ops.mesh.primitive_cylinder_add(
        radius=tube_radius * 0.95, depth=HST_SIZE * 0.18,
        vertices=32, location=(tube_length / 2.0 + HST_SIZE * 0.09, 0, 0))
    cap = bpy.context.active_object
    cap.rotation_euler = (0.0, math.radians(90), 0.0)
    bpy.ops.object.transform_apply(rotation=True)
    _add_part(cap, "HST_ApertureCap", aperture_mat)

    # ── Aft shroud (slightly wider, shorter cylinder at -X end) ───────────────
    bpy.ops.mesh.primitive_cylinder_add(
        radius=tube_radius * 1.12, depth=HST_SIZE * 1.6,
        vertices=32,
        location=(-tube_length / 2.0 + HST_SIZE * 0.35, 0, 0))
    aft = bpy.context.active_object
    aft.rotation_euler = (0.0, math.radians(90), 0.0)
    bpy.ops.object.transform_apply(rotation=True)
    _add_part(aft, "HST_AftShroud", body_mat)
    bpy.ops.object.shade_smooth()

    # ── Solar array wings ─────────────────────────────────────────────────────
    # Two pairs of panels (fore and aft of the mid-point), one on each side (+Z / -Z).
    # Each panel is a flat box (cube scaled thin).
    panel_w  = HST_SIZE * 4.8   # width along X (telescope axis)
    panel_h  = HST_SIZE * 2.4   # height along Y
    panel_th = HST_SIZE * 0.06  # thickness along Z

    # Spanwise offset from tube centre (Z direction)
    panel_z_offset = tube_radius + panel_h * 0.50 + HST_SIZE * 0.05

    # Axial offset: panels sit roughly at X = ±1.2 * HST_SIZE
    panel_x_offset = HST_SIZE * 1.2

    for side_name, z_sign in (("Port", 1.0), ("Starboard", -1.0)):
        bpy.ops.mesh.primitive_cube_add(size=1.0, location=(0, 0, 0))
        panel = bpy.context.active_object
        panel.scale = (panel_w, panel_h, panel_th)
        bpy.ops.object.transform_apply(scale=True)
        panel.location = (panel_x_offset, 0.0, z_sign * panel_z_offset)
        _add_part(panel, f"HST_SolarPanel_{side_name}", panel_mat)

    # ── High-gain antenna dishes (two, fore and aft on opposite sides) ────────
    dish_r      = HST_SIZE * 0.38
    dish_depth  = HST_SIZE * 0.12
    boom_r      = HST_SIZE * 0.05
    boom_len    = HST_SIZE * 0.55

    antenna_specs = [
        # (x_pos, y_sign)  — one on +Y side near front, one on -Y near aft
        ( tube_length * 0.28,  1.0),
        (-tube_length * 0.28, -1.0),
    ]
    for idx, (ax, y_sign) in enumerate(antenna_specs):
        # Boom (thin cylinder along Y)
        bpy.ops.mesh.primitive_cylinder_add(
            radius=boom_r, depth=boom_len, vertices=12,
            location=(ax, y_sign * (tube_radius + boom_len / 2.0), 0.0))
        boom = bpy.context.active_object
        # no rotation needed — cylinder Z aligns with Y after we rotate 90° X
        boom.rotation_euler = (math.radians(90), 0.0, 0.0)
        bpy.ops.object.transform_apply(rotation=True)
        _add_part(boom, f"HST_AntennaBoom_{idx}", body_mat)

        # Dish (flat cylinder acting as parabolic dish face)
        bpy.ops.mesh.primitive_cylinder_add(
            radius=dish_r, depth=dish_depth, vertices=24,
            location=(ax, y_sign * (tube_radius + boom_len + dish_depth / 2.0), 0.0))
        dish = bpy.context.active_object
        dish.rotation_euler = (math.radians(90), 0.0, 0.0)
        bpy.ops.object.transform_apply(rotation=True)
        _add_part(dish, f"HST_AntennaDish_{idx}", body_mat)

    # ── Fine-guidance sensor bays (small rectangular boxes on tube equator) ───
    # Three bays at 90° intervals around the tube (Hubble has 3 FGS units).
    fgs_w = HST_SIZE * 0.55
    fgs_h = HST_SIZE * 0.40
    fgs_d = HST_SIZE * 0.30
    fgs_x = HST_SIZE * 0.5   # axial position (near front third)
    fgs_r = tube_radius + fgs_d / 2.0

    for bay_idx, angle_deg in enumerate([60.0, 180.0, 300.0]):
        ang = math.radians(angle_deg)
        bpy.ops.mesh.primitive_cube_add(size=1.0, location=(0, 0, 0))
        bay = bpy.context.active_object
        bay.scale = (fgs_w, fgs_h, fgs_d)
        bpy.ops.object.transform_apply(scale=True)
        bay.location = (fgs_x, math.sin(ang) * fgs_r, math.cos(ang) * fgs_r)
        _add_part(bay, f"HST_FGS_Bay_{bay_idx}", body_mat)

    # ── Animate orbit & slow axial roll ──────────────────────────────────────
    animate_local_orbit(orbit_empty, ORBIT_RADIUS, ORBIT_PERIOD)

    # HST keeps its solar panels roughly sun-pointing — simulate with slow roll
    animate_pure_z_spin(hst_root, ORBIT_PERIOD * 3, direction=1)

    print("  ✅ Hubble Space Telescope built (tube + solar arrays + antennas + FGS bays)")
    return hst_root

def animate_orbit(planet_obj, orbit_radius, period_frames, scale_factor=1.0):
    scene = bpy.context.scene
    total_frames   = scene.frame_end
    orbit_duration = period_frames * scale_factor
    steps          = 64
    num_keys       = int(total_frames / orbit_duration * steps) + steps
    for k in range(num_keys + 1):
        frame = 1 + k * (orbit_duration / steps)
        if frame > total_frames:
            break
        angle = (2 * math.pi * k) / steps
        planet_obj.location = (
            math.cos(angle) * orbit_radius,
            math.sin(angle) * orbit_radius,
            0,
        )
        planet_obj.keyframe_insert(data_path="location", frame=frame)
    if planet_obj.animation_data and planet_obj.animation_data.action:
        for fcurve in planet_obj.animation_data.action.fcurves:
            for kf in fcurve.keyframe_points:
                kf.interpolation = "LINEAR"

def animate_pure_z_spin(obj, period_frames, direction=1):
    scene        = bpy.context.scene
    total_frames = scene.frame_end
    obj.rotation_mode  = "XYZ"
    obj.rotation_euler = (0.0, 0.0, 0.0)
    num_spins = max(1, math.ceil(total_frames / period_frames))
    for spin in range(num_spins + 1):
        frame = 1 + spin * period_frames
        obj.rotation_euler[2] = direction * 2 * math.pi * spin
        obj.keyframe_insert(data_path="rotation_euler", index=2, frame=frame)
    if obj.animation_data and obj.animation_data.action:
        for fcurve in obj.animation_data.action.fcurves:
            if fcurve.array_index == 2:
                for kf in fcurve.keyframe_points:
                    kf.interpolation = "LINEAR"
                fcurve.modifiers.new(type="CYCLES")

def build_solar_system():
    clear_scene()
    setup_render_settings()

    print("\n📁 Checking textures...\n")
    for body, path in TEXTURE_URLS.items():
        status = "✅" if os.path.exists(path) else "❌ MISSING"
        print(f"  {status}  {body}: {path}")
    print()

    # ── Sun ──────────────────────────────────────────────────────────────────
    sun_mat  = make_sun_material("Sun_Mat", TEXTURE_URLS.get("Sun"))
    sun_tilt = make_tilt_empty("Sun_TiltEmpty", AXIAL_TILTS.get("Sun", 0.0))
    sun_obj  = add_sphere("Sun", SUN_RADIUS, (0, 0, 0), sun_mat)
    sun_obj.parent = sun_tilt
    animate_pure_z_spin(sun_obj, SUN_ROTATION_PERIOD)

    add_starfield()

    # ── Planets ───────────────────────────────────────────────────────────────
    compressed_periods = {p[0]: math.log(p[3]) for p in PLANETS}
    max_compressed     = max(compressed_periods.values())
    total_frames       = bpy.context.scene.frame_end
    orbit_scale        = total_frames / max_compressed

    earth_tilt_empty = None

    for (name, size, orbit_r, period, fallback_color) in PLANETS:
        add_orbit_ring(f"{name}_Orbit", orbit_r)

        tex_path  = TEXTURE_URLS.get(name)
        mat       = make_planet_material(name, tex_path, fallback_color)
        direction = ROTATION_DIRECTION.get(name, 1)
        tilt_deg  = AXIAL_TILTS.get(name, 0.0)

        if name == "Saturn":
            bpy.ops.object.empty_add(type="PLAIN_AXES", location=(orbit_r, 0, 0))
            orbit_pivot = bpy.context.active_object
            orbit_pivot.name               = "Saturn_OrbitPivot"
            orbit_pivot.empty_display_size = 0.01

            tilt_empty = make_tilt_empty("Saturn_TiltEmpty", tilt_deg, parent=orbit_pivot)
            planet     = add_sphere(name, size, (0, 0, 0), mat)
            planet.parent = tilt_empty

            create_saturn_rings(orbit_pivot, planet_radius=size,
                                texture_path=TEXTURE_URLS.get("Saturn_Ring"))

            animate_orbit(orbit_pivot, orbit_r,
                          compressed_periods[name], scale_factor=orbit_scale)
            animate_pure_z_spin(planet, ROTATION_PERIODS.get(name, 60), direction=direction)

        else:
            tilt_empty          = make_tilt_empty(f"{name}_TiltEmpty", tilt_deg)
            tilt_empty.location = (orbit_r, 0, 0)
            planet              = add_sphere(name, size, (0, 0, 0), mat)
            planet.parent       = tilt_empty

            animate_orbit(tilt_empty, orbit_r,
                          compressed_periods[name], scale_factor=orbit_scale)
            animate_pure_z_spin(planet, ROTATION_PERIODS.get(name, 60), direction=direction)

            if name == "Earth":
                earth_tilt_empty = tilt_empty

    # ── Moon ─────────────────────────────────────────────────────────────────
    if earth_tilt_empty is not None:
        build_moon(earth_tilt_empty)
    else:
        print("  ⚠️ Earth tilt empty not found — Moon skipped.")

    # ── Hubble Space Telescope ────────────────────────────────────────────────
    if earth_tilt_empty is not None:
        build_hubble(earth_tilt_empty)
    else:
        print("  ⚠️ Earth tilt empty not found — Hubble skipped.")

    # ── Asteroid Belt ─────────────────────────────────────────────────────────
    build_asteroid_belt(
        num_asteroids=500,
        inner_r=25.5,
        outer_r=28.0,
        max_z=0.6,
        seed=42,
        num_rogues=20
    )

    # ── Kuiper Belt ──────────────────────────────────────────────────────────
    build_kuiper_belt(
        num_kbos=700,
        inner_r=68.0,
        outer_r=88.0,
        max_z=3.0,
        seed=99,
        num_rogues=35
    )

    setup_camera()
    setup_lighting()
    force_rendered_viewport()
    bpy.context.scene.frame_set(1)

    print("\n✅  Done! Press F12 to render. Press SPACE to animate.")
    print("    🪐 Saturn rings tilted 26.7° for full visibility.")
    print("    🔭 Hubble Space Telescope: 28.5° inclined orbit (r=1.6) around Earth.")
    print("       MLI body + deep-blue solar arrays + HGA dishes + FGS bays.")
    print("    🪨 Asteroid belt: 500 rocks (r=25.5-28.0) + 20 rogues.")
    print("    ❄️  Kuiper Belt:   700 KBOs (r=68-88) + 35 scattered-disc objects.")
    print("       Icy blues + reddish tholins, emission=2.5, camera clip=1000,")
    print("       larger bodies (0.12-0.85), dense inner + sparse outer ring.")


build_solar_system()
